\section{GETRF}
\begin{frame}
\frametitle{GETRF}
\begin{itemize}
    \item GEneral matrix TRiangular Factorization
    \item Solve systems of linear equations of the form \textit{Ax=b}
    \item Breaks a matrix A into 2 triagular matrices L and U
\end{itemize}
\begin{figure}
    \centering
    \includegraphics[width=0.8\textwidth]{A=LU.png}
    \caption{LU Factorization}
\end{figure}
\end{frame}

\begin{frame}[fragile]
\frametitle{GETRF blocked}
\begin{columns}[T]

    \begin{column}{0.5\textwidth}
        \centering
        \includegraphics[width=\textwidth]{GETRF_BLOCKED.png}
        \vfill
        \small{Figure: Blocked version of GETRF}
    \end{column}

    \begin{column}{0.5\textwidth}
\begin{lstlisting}[basicstyle=\tiny\ttfamily]
for k = 0 to small_dim step block_size:
    # 1. Factorize the diagonal block
    GETRF_SCALAR( A[k,k] )

    # 2. Update the "Panel" Row
    TRSM_LOWER( A[k,k], A[k, k+size] )

    # 3. Update the "Panel" Column
    TRSM_UPPER( A[k,k], A[k+size, k] )

    # 4. Update the Remaining Matrix
    GEMM( A[k+size, k], A[k, k+size]
          -> A[k+size, k+size] )
\end{lstlisting}
\end{column}

\end{columns}
\end{frame}

\section{OpenMP}
\begin{frame}
\frametitle{Concept}
\begin{itemize}
    \item Use several cores on a machine
    \item Abstraction and simplification of the use of threads
    \item Split the computation between threads to speed up computation
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{DGEMM OpenMP}
\begin{itemize}
    \item Use the code from the sequential version
    \item Parallelize the micro kernel
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{DGEMM OpenMP}

\begin{frame}
\frametitle{DGEMM OpenMP tiled}
\begin{itemize}
    \item Use the base code composed of 3 imbricated loops
    \item Parallelize the 2 outer loops
    \item \textit{\#pragma omp parallel for collapse(2) firstprivate(mm,nn,kk,lbeta,k)}
\end{itemize}
\end{frame}
