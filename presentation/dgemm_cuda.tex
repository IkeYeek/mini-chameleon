\begin{frame}{Hardware Feature: NVIDIA Tensor Cores}
    \begin{columns}
        \begin{column}{0.6\textwidth}
            \begin{itemize}
                \item \textbf{The Operation:} $D = A \times B + C$
                \item \textbf{Block-Level Computing:}
                \begin{itemize}
                    \item Unlike CUDA cores (scalar/vector), Tensor Cores operate on \textbf{entire matrices} (blocks)
                    \item Ampere is the first generation with native \textbf{$8 \times 8 \times 4$ FP64} support
                \end{itemize}
                \item \textbf{Efficiency:} Dramatically reduces instruction overhead and maximizes throughput
            \end{itemize}
        \end{column}
        \begin{column}{0.4\textwidth}
            \begin{center}
                \includegraphics[width=\textwidth]{assets/tensor_core_illustration.png}
                \captionof{figure}{Hardware-native Block MMA}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Abstractions with NVIDIA CuTe}
    \begin{itemize}
        \item \textbf{The Problem:} Manual indexing for tiled matrix multiplication is error-prone and rigid
        \item \textbf{The Solution:} \textbf{CuTe} (part of CUTLASS 3.0+)
        \item \textbf{Key Concepts:}
        \begin{itemize}
            \item \textbf{Layouts:} Decouple data coordinates from physical memory offsets
            \item \textbf{Tensors:} Multi-dimensional views of data using layouts
            \item \textbf{Tiled Operations:} High-level abstractions for \textbf{Copy} and \textbf{MMA} (Matrix-Multiply-Accumulate)
        \end{itemize}
        \item \textbf{Benefit:} Allows us to focus on the algorithmic tiling strategy rather than pointer arithmetic
    \end{itemize}
\end{frame}

\begin{frame}{Optimization: caching policy}
    \begin{itemize}
        \item \textbf{Problem:} When looking at the data loading scheme, we notice that each block is loaded from global memory once. We are misusing our L1 cache
        \item \textbf{Solution:} Instruct the hardware to skip L1 when loading data from global memory
    \end{itemize}
\end{frame}
